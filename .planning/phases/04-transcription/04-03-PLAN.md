    ```typescript
      speaker_id: string;
      spoken_at: string;   // ISO string
      debate_time_mmss: string;
      text: string;
      confidence_score: number | null;
    }
    ```

    ```sql
    SELECT
      te.id,
      te.speaker_id,
      te.spoken_at,
      te.debate_time_mmss,
      te.text,
      te.confidence_score
    FROM listening.transcript_entries te
    WHERE te.debate_id = $1
      AND ($2::timestamptz IS NULL OR te.spoken_at < $2)
    ORDER BY te.spoken_at ASC
    LIMIT $3
    ```

    **Part A: hooks/useTranscriptSync.ts**

    Subscribes to the Supabase Realtime broadcast channel `transcript-{debateId}`.

    IMPORTANT: Broadcast is EPHEMERAL. Any messages sent before subscription are lost.
    The correct pattern (matching Pitfall 5 in research): load DB snapshot FIRST via the GET endpoint in TranscriptPanel's useEffect, THEN subscribe to the broadcast channel. useTranscriptSync itself only handles the subscription, not the initial load.

    ```typescript
    'use client';
    import { useEffect, useCallback } from 'react';
    import { getSupabaseBrowserClient } from '@/lib/supabase/client';

    export interface FinalEntry {
      speakerId: string;
      text: string;
      spokenAt: string;         // ISO timestamp string
      debateTimeMmss: string;
    }

    export interface InterimEntry {
      speakerId: string;
      text: string;
    }

    export function useTranscriptSync(
      debateId: string,
      onFinal: (entry: FinalEntry) => void,
      onInterim: (entry: InterimEntry) => void,
    ): void {
      // Stable callbacks to avoid re-subscribing on every render
      const stableFinal = useCallback(onFinal, []);    // eslint-disable-line react-hooks/exhaustive-deps
      const stableInterim = useCallback(onInterim, []); // eslint-disable-line react-hooks/exhaustive-deps

      useEffect(() => {
        const supabase = getSupabaseBrowserClient();

        const channel = supabase
          .channel(`transcript-${debateId}`)
          .on('broadcast', { event: 'final' }, (payload) => {
            stableFinal(payload.payload as FinalEntry);
          })
          .on('broadcast', { event: 'interim' }, (payload) => {
            stableInterim(payload.payload as InterimEntry);
          })
          .subscribe();

        return () => { void supabase.removeChannel(channel); };
      }, [debateId, stableFinal, stableInterim]);
    }
    ```

    **Part C: components/transcript/TranscriptPanel.tsx**

    Full live-updating panel. Loads initial entries from DB on mount, then subscribes to broadcast. Renders segment header dividers between LD segments.

    Props:
    ```typescript
    interface Speaker {
      displayName: string;
      role: 'affirmative' | 'negative' | 'moderator';
    }

    interface DebateSegment {
      id: string;
      name: string;
      actual_start: string | null;  // ISO timestamp or null if not yet started
      duration_seconds: number;
    }

    interface TranscriptPanelProps {
      debateId: string;
      speakers: Record<string, Speaker>;  // speakerId (UUID) → speaker info
      segments: DebateSegment[];          // ordered array from debate store
    }
    ```

    Behavior:
    1. On mount: fetch `/api/debates/${debateId}/transcript` → setEntries(result.entries)
    2. Subscribe via useTranscriptSync:
       - onFinal: append to entries; remove interim for that speakerId; auto-scroll if following live
       - onInterim: update interims[entry.speakerId] = entry.text
    3. Auto-scroll: `isFollowingLive` ref (starts true). Scroll event: set to atBottom < 100px.
    4. "Back to live" button: absolutely positioned bottom-right of panel, visible only when !isFollowingLive; on click sets isFollowingLive=true and scrolls to bottom.
    5. Pre-debate placeholder: if entries.length === 0 and no interims, show:
       `<p className="text-slate-500 text-sm">Transcript will appear here when speakers begin.</p>`
    6. Interims: render at bottom in italic, lighter color (`text-slate-400 italic text-sm`), keyed by speakerId.

    Segment header rendering:
    - Add a helper: `function formatDuration(seconds: number): string` that formats as "m:ss" (e.g., 480 → "8:00", 65 → "1:05").
    - When rendering the entries list, group entries under their segment by comparing each entry's `spoken_at` timestamp against `segment.actual_start` timestamps. An entry belongs to the latest segment whose `actual_start` is <= the entry's `spoken_at`.
    - For each segment that has at least one entry (or is the current active segment), render a header row BEFORE the first entry in that segment:
      ```tsx
      <div className="py-2 px-2 mt-3 mb-1 border-b border-slate-700">
        <span className="text-slate-400 text-xs font-semibold uppercase tracking-wide">
          {segment.name} — {formatDuration(segment.duration_seconds)}
        </span>
      </div>
      ```
    - Segments with `actual_start: null` have not started yet — skip their headers entirely.
    - This produces the locked UX: "Affirmative Constructive — 8:00" dividers between each LD segment.
